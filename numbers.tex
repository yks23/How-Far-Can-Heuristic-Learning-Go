% ===========================================================================
% numbers.tex -- every quantitative claim in the paper, defined in one place.
%
% The body never writes a literal number. It writes \val{key}, and the value
% lives here. Two reasons: a number that appears in the abstract, a table, and
% the discussion can only disagree with itself if it is typed three times; and
% a provisional number has to be visibly provisional until it is measured.
%
%   \defval{key}{value}   a measured number. Prints plain.
%   \defTBD{key}{guess}   not measured yet. Prints in accent bold while
%                         \hldraft is on, so it cannot ship unnoticed, and is
%                         listed by \printTBD.
%   \val{key}             use a number. An undefined key prints a red marker
%                         and raises a LaTeX warning rather than failing quietly.
%   \todo{note}           a prose-level gap, same visibility rules.
%   \printTBD             renders the outstanding-measurement list. Put it at
%                         the end of the document; it disappears with \hldraft.
%
% Camera-ready: \togglefalse{hldraft}. Any key still declared with \defTBD then
% raises a hard \PackageError naming the key, and the build fails. This used to
% be a procedural rule -- "\printTBD must read empty before that switch is
% thrown" -- which put the paper one forgotten check away from asserting "XX%"
% as a measured value, since flipping the toggle removes the accent colour and
% \printTBD in the same stroke. It is now enforced by the macro. To ship, every
% \defTBD must have become a \defval; there is no override, by design.
% ===========================================================================

\newtoggle{hldraft}
\toggletrue{hldraft}

\newcommand{\hlnumtbdlist}{}

\newcommand{\defval}[2]{\csdef{hlnum@#1}{#2}\csundef{hlntbd@#1}}
\newcommand{\defTBD}[2]{%
  \csdef{hlnum@#1}{#2}\csdef{hlntbd@#1}{}\listadd{\hlnumtbdlist}{#1}}

% A still-TBD key used with \hldraft off is a hard error, not a silent
% substitution. Turning the toggle off is the camera-ready action, so this is
% the moment a leftover placeholder would stop being visible: it loses its
% accent colour AND \printTBD disappears in the same stroke, taking with it the
% only list that would have reported it. Erroring here means the build refuses
% rather than quietly shipping "XX%" as a measured quantity. There is
% deliberately no override -- if placeholders are wanted, leave \hldraft on.
\newcommand{\val}[1]{%
  \ifcsdef{hlnum@#1}{%
    \ifcsdef{hlntbd@#1}{%
      \iftoggle{hldraft}{\textcolor{brandaccent}{\bfseries\csuse{hlnum@#1}}}%
        {\PackageError{hlnumbers}{Unmeasured key '#1' used with hldraft off}%
          {Key '#1' is still \string\defTBD. Measure it and convert it to
           \string\defval, or re-enable \string\toggletrue{hldraft}. The
           camera-ready build will not proceed with placeholders in it.}%
         \csuse{hlnum@#1}}%
    }{\csuse{hlnum@#1}}%
  }{%
    \textcolor{red}{\bfseries[?\,\texttt{#1}]}%
    \PackageWarning{hlnumbers}{Undefined number key '#1'}%
  }%
}

\newcommand{\todo}[1]{%
  \iftoggle{hldraft}{\textcolor{red}{\bfseries[TODO: #1]}}{}}

\newcommand{\hlnumtbditem}[1]{\item \texttt{#1} \dotfill\ \csuse{hlnum@#1}}
\newcommand{\printTBD}{%

    empty before submission.
    \begin{itemize}[nosep,leftmargin=*]
      \forlistloop{\hlnumtbditem}{\hlnumtbdlist}
    \end{itemize}}{}}

% ---------------------------------------------------------------------------
% Sample efficiency against RL. The crossover matters more than the ratio: the
% claim is budget-conditional, not a claim that HL dominates. loseaxis names
% the denominator on which HL is the worse method, stated up front on purpose.
%
% Both keys are measured against MODEL-BASED RL, the primary baseline, and the
% text must say so wherever they appear -- policy gradient will cross over
% somewhere else, and a bare "RL overtakes at crossover" invites the reader to
% assume whichever baseline is least favourable to us. If the two baselines cross
% at materially different budgets, that is a result worth reporting and needs its
% own key rather than a hedge on this one.
% ---------------------------------------------------------------------------
\defTBD{effratio}{XX$\times$}
\defTBD{crossover}{$\sim$XXX k}

% ---------------------------------------------------------------------------
% The ceiling. Frontier-Eng (arXiv 2604.12290) reports improvement frequency
% decaying as t^-1 with a plateau by 50-100 iterations on single-agent tasks.
% Whether an adversarial setting decays slower is the interesting comparison,
% so the exponent is a measured quantity here, not an assumed one.
% ---------------------------------------------------------------------------
\defTBD{decay}{$t^{-\alpha}$}
\defTBD{plateau}{$\sim$XX}
% Prose-valued keys stay English-only: the document is compiled with pdfLaTeX,
% which cannot typeset CJK without CJKutf8, and \printTBD renders every
% registered key whether the body cites it or not.
% Named rather than left blank: the hypothesis is that the plateau is the
% agent's maintenance capacity, measured as maintcap. Still TBD because it is a
% hypothesis about a mechanism, not yet a measured result -- and the honest
% failure mode is that the plateau tracks the game instead of the agent, which
% maintcapspread is what distinguishes. Reads as a noun phrase so it can follow
% "approaches" or "we trace the ceiling to".
\defTBD{ceilingcause}{the agent's maintenance capacity}
\defTBD{ntransfer}{X}

% ---------------------------------------------------------------------------
% The mechanism behind the ceiling: maintenance capacity.
%
% The hypothesis is that HL plateaus when the program's description length
% approaches what the agent can still revise without regression -- so the
% ceiling is a property of the maintainer, not of the paradigm, and moves with
% model capability. Three quantities make that testable rather than rhetorical.
%
% maintcap is the load-bearing one. If it converges to a similar magnitude
% across games of very different complexity, the ceiling is the agent's; if it
% tracks the game instead, it is the game's. Either result is publishable, and
% they are distinguishable, which is the point of measuring it.
%
% Convergence alone is not enough, though, and reading it as sufficient is the
% trap here. The mechanism assigns the capacity to the agent and K(pi_good) to the
% game, so it predicts a *pair*: maintcap converges across games while the Elo
% reached at that plateau varies with the game. If both converge, the more likely
% explanation is that the games do not differ much in K(pi_good) -- in which case
% the cross-game comparison is uninformative rather than confirmatory. Pair this
% key with Elo-at-plateau on the same games whenever it is reported.
%
% Units matter: report compressed bytes, not lines. Lines are a style artifact
% and are not comparable across the languages the archive contains.
%
% There is an external anchor to check this against. longcontextbugfix2026 finds
% successful agentic repair trajectories staying under 20-30k tokens on
% SWE-bench Verified, with resolve rates falling past that. Their axis is
% accumulated context, not program description length, so this is an
% order-of-magnitude sanity check after unit conversion and nothing stronger.
% Landing near their range is convergent evidence worth naming; landing far from
% it is a finding about how the two axes differ, not a measurement error. Decide
% the conversion before looking at our data, or the check is worthless.
% ---------------------------------------------------------------------------
\defTBD{maintcap}{$\sim$XX kB}
\defTBD{maintcapspread}{$\pm$XX\%}
% The second half of that pair, and the reason maintcapspread is interpretable.
% The spread of Elo reached at the plateau across the same games. The mechanism
% wants this one WIDE while maintcapspread is narrow: same maintainable size,
% different strength at that size, because K(pi_good) differs. A narrow value
% here is the outcome that costs the cross-game argument, so report it even --
% especially -- if it comes out narrow.
\defTBD{eloplatspread}{$\pm$XX}

% Kolmogorov complexity is uncomputable. It motivates the axis; it is never
% measured. What is measured is a compression proxy, and the proxy has to be
% named in the text every time the axis is used, or a referee will and should
% object. This key holds the proxy's name so the paper cannot drift between
% two of them.
\defTBD{kproxy}{compressed size of a reference implementation}

% The archive's free experiment: for each Elo band, the shortest program that
% reaches it (lower envelope, not mean -- description length is a minimum).
% A slope near zero means the paradigm has room at that skill level; a knee
% means it is running out. Games are ordered on this axis, not on rule length:
% Go has short rules and an incompressible policy, which is why rule
% complexity is the wrong x-axis and this is the right one.
\defTBD{envslope}{$+$XX bytes per 100 Elo}
\defTBD{envknee}{$\sim$XXXX Elo}
